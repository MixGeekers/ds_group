89: // DSH commits the assistant message before tool results exist. A tail that still
90: // owes a result is dropped whole: an adapter rejects a tool call with no result,
91: // and an aside does not wait for one or invent one.
92: export function trimToolBoundary(messages) {
93:   if (!Array.isArray(messages) || messages.length === 0) return []
94:   let index = messages.length
95:   while (index > 0 && messages[index - 1]?.role === 'tool') index -= 1
96:   const calls = callIds(messages[index - 1])
97:   if (calls.length === 0) return messages
98:   const answered = new Set()
99:   for (const message of messages.slice(index)) {
100:     if (typeof message?.toolCallId === 'string') answered.add(message.toolCallId)
101:   }
102:   const balanced = calls.every(id => typeof id === 'string' && answered.has(id))
103:   return balanced ? messages : messages.slice(0, index - 1)
104: }

155: export async function callAside(ctx, rt, snapshot, { extra = [], question, promptFor, kind, retry = false, signal }) {
156:   const messages = rt.tagUsage([
157:     ...snapshot.messages,
158:     ...extra,

160:   ], { sessionId: snapshot.sessionId, kind })
161:   const assembler = new BlockAssembler()
162:   const tools = snapshot.header?.tools
163:   for await (const chunk of ctx.llm.stream({
164:     ...snapshot.header?.config,
165:     messages,
166:     toolHistory: snapshot.toolHistory,

168:     sessionId: snapshot.sessionId,
169:     signal,
170:   })) assembler.push(chunk)
171:   const finish = assembler.finish
172:   if (finish.kind === 'error' || finish.kind === 'aborted') {
173:     return { failed: finish.failure?.message ?? finish.kind }
174:   }

177:   if (finish.kind === 'max-tokens') return { text, cutOff: true }
